PM2 — Quick Commands

รันจาก macOS GUI Terminal เท่านั้น (LNP)

# Status
pm2 status
pm2 list

# Logs
pm2 logs api-server --lines 50
pm2 logs mqtt-subscriber --lines 30
pm2 logs alert-worker --lines 20

# Restart
open -a Terminal /Users/dojojin/vigil-platform/scripts/pm2-lan-safe-restart.command

# Restart service เดียว (จาก GUI Terminal เท่านั้น)
pm2 restart api-server
pm2 restart alert-worker

# Full lifecycle
bash scripts/services.sh start
bash scripts/services.sh stop
bash scripts/services.sh status

Docker — Infrastructure

# Status
docker ps
docker compose ps

# Logs
docker logs vigil-postgres --tail 30
docker logs vigil-emqx --tail 30

# Restart docker services
docker compose restart
docker compose up -d --force-recreate   # ถ้า port binding ผิด

# Database shell
docker exec -it vigil-postgres psql -U vigil_sql -d vigil_platform

# EMQX UI
open http://localhost:18083   # admin / ดูรหัสผ่านใน .env → EMQX_DASHBOARD_PASSWORD

Database — Backup & Restore

# Manual backup
./scripts/backup.sh
# → backups/vigil_platform_YYYY-MM-DD_hhmmss.dump

# ดู backup list
ls -lh backups/

# Restore (ต้องพิมพ์ "yes" ยืนยัน)
./scripts/restore.sh backups/vigil_platform_2026-05-29_030000.dump

# ดู TOC ของ backup
pg_restore --list backups/vigil_platform_latest.dump | head -30

# Check schema migrations applied
docker exec -it vigil-postgres psql -U vigil_sql -d vigil_platform \
  -c "SELECT filename, applied_at FROM schema_migrations ORDER BY applied_at DESC LIMIT 10;"

Auto-backup ด้วย launchd

ตั้งครั้งเดียว: cp scripts/com.dojojin.dashboard.backup.plist ~/Library/LaunchAgents/ && launchctl load -w ~/Library/LaunchAgents/com.dojojin.dashboard.backup.plist

Cloudflare Tunnel

# ตรวจสถานะ tunnel (local)
curl -s http://localhost:3000/api/health | node -e "const d=require('fs').readFileSync('/dev/stdin','utf8'); console.log(JSON.parse(d).status)"

# ตรวจผ่าน public URL
curl -I https://dashboard.dojojin.tech/api/health

# Restart tunnel (launchd)
launchctl stop com.cloudflare.cloudflared
launchctl start com.cloudflare.cloudflared

# ดู tunnel log
tail -f /Library/Logs/com.cloudflare.cloudflared.out.log
tail -f /Library/Logs/com.cloudflare.cloudflared.err.log

Schema Migrations

# Apply migrations manually (นอกจาก api-server boot)
cd src && npm run migrate

# ดู migrations ที่ apply แล้ว
docker exec -it vigil-postgres psql -U vigil_sql -d vigil_platform \
  -c "SELECT filename, applied_at FROM schema_migrations ORDER BY filename;"

# ดู migration files ทั้งหมด
ls db/db_migration_*.sql | sort

Migration fail = api-server ไม่ start

ห้าม comment out migration ที่ fail — restore backup แล้วแก้ SQL ให้ถูก (GOTCHAS #19)

Health Check

# API health endpoint
curl http://localhost:3000/api/health
curl http://localhost:3000/api/health/details   # รายละเอียดทุก component

# ตรวจ disk
df -h /Users/dojojin
du -sh backups/

# ตรวจ memory
pm2 status   # ดู memory column
top -pid $(pm2 id api-server | head -1)
