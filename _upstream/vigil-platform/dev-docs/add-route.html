Recipe: เพิ่ม REST Route

Vigil Platform ใช้ Express 5. Routes ขนาดใหญ่แยกเป็น module ใน src/routes/ ด้วย factory pattern; routes เล็กหรือ one-off ใส่ตรงใน src/api-server.js ได้

ทำไม S4 ถึงแยก api-server.js → 19 route files (MAINT-2T-001)

  • อ่านได้ทันที — ไฟล์ 80–1,200 ลบ. เห็น context ทั้งหมดในหน้าเดียว vs api-server.js เดิม 7,156 ลบ.
  • AI context ใช้งานจริง — โหลดเฉพาะ src/routes/line.js เมื่อแก้ LINE แทนที่จะโหลดทั้ง 7k บรรทัด
  • git diff อ่านออก — PR ที่แตะ routes/cameras.js = รู้ทันทีว่าแตะกล้องเท่านั้น
  • test แยกได้ — factory function รับ (app, pool, deps) → inject mock pool โดยไม่ boot Express ทั้งก้อน
  • แก้ bug ไม่กระทบส่วนอื่น — แก้ stats ไม่มีทางพลาดแตะ LINE หรือ auth โดยบังเอิญ
  • Runtime ไม่เปลี่ยน — require() 19 ไฟล์แทบไม่ต่างจาก 1 ไฟล์ใหญ่

ตัดสินใจก่อน: inline หรือ route module?

กรณีวิธี
endpoint เดียว ไม่ซับซ้อน ไม่เข้า module ไหน เพิ่มตรงใน src/api-server.js ต่อท้าย require block ที่เกี่ยวข้อง
feature ใหม่ที่มีหลาย endpoint หรือ logic ซับซ้อน สร้าง src/routes/<name>.js factory module (แนะนำ)

Option A: เพิ่มตรงใน api-server.js

หลัง S4 split (2026-06-14) ส่วนใหญ่ route อยู่ใน src/routes/ แล้ว — Option A ใช้สำหรับ one-off endpoint ที่ไม่เข้า module ไหน ค้น require('./routes/ ใน api-server.js แล้วเพิ่ม route ต่อท้าย require block ที่เกี่ยวข้อง:

app.get('/api/my-feature', auth.requireAuth, async (req, res) => {
  try {
    const { rows } = await pool.query('SELECT ...', [req.query.param]);
    res.json(rows);
  } catch (err) {
    routeError(res, err, 'GET /api/my-feature');
  }
});

routeError

routeError ถูก require ไว้ที่ต้นไฟล์แล้ว (src/helpers/routeError.js) — ใช้ได้เลย ไม่ต้อง import เพิ่ม

Option B: สร้าง Route Module ใหม่

1

สร้างไฟล์ src/routes/<feature-name>.js — ใช้ factory pattern ตามนี้:

// ============================================================
// Vigil Platform — Routes: <Feature Name>
// @author Prakasit Rochanavipart (Dojo-mAn)
// @copyright (c) 2025-2026 Prakasit Rochanavipart. All Rights Reserved.
// @license Proprietary
// ============================================================
'use strict';

const routeError = require('../helpers/routeError');

module.exports = function myFeatureRoutes(app, pool, { auth }) {

  // GET /api/my-feature
  app.get('/api/my-feature', auth.requireAuth, async (req, res) => {
    try {
      const { rows } = await pool.query(
        'SELECT * FROM my_table WHERE active = $1',
        [true]
      );
      res.json(rows);
    } catch (err) {
      routeError(res, err, 'GET /api/my-feature');
    }
  });

  // POST /api/my-feature
  app.post('/api/my-feature', auth.requireAdmin, async (req, res) => {
    const { name } = req.body;
    if (!name) return res.status(400).json({ error: 'name required' });
    try {
      const { rows } = await pool.query(
        'INSERT INTO my_table (name) VALUES ($1) RETURNING *',
        [name]
      );
      res.json(rows[0]);
    } catch (err) {
      routeError(res, err, 'POST /api/my-feature');
    }
  });

};
2

Register ใน src/api-server.js — ค้น require('./routes/ เพื่อหาตำแหน่งที่เหมาะ แล้วเพิ่ม:

require('./routes/my-feature')(app, pool, { auth });

ถ้า route ต้องการ dep เพิ่ม เช่น getIP หรือ invalidateSystemSetting — ส่งผ่าน deps object:

require('./routes/my-feature')(app, pool, { auth, getIP });

Auth Gate — เลือกระดับให้ถูก

Middlewareใครใช้เมื่อ
auth.requireAuth viewer, admin, auditor อ่านข้อมูล — events, cameras, stats, appearances
auth.requireAdmin admin เท่านั้น เขียน / ลบ / จัดการ config — alert rules, settings, users, LINE
auth.requireAdminOrAuditor admin + auditor อ่าน + export — report history, audit log
ไม่มี middleware ทุกคน (public) branding, health ping, login endpoint เท่านั้น

ข้อควรระวัง

  • Auditor write-blocking เป็น global middleware ใน api-server.js — ไม่ต้องเพิ่ม check เองใน route
  • Internal Puppeteer calls ผ่าน req.internal === true — requireAuth/Admin/AdminOrAuditor honor ค่านี้อยู่แล้ว
  • อย่าอ่าน X-Forwarded-For ตรงๆ — ใช้ getIP(req) แทน (GOTCHAS #38)

Token-Pre-Check Gate (สำหรับ push receiver — ไม่ใช้ session)

Route ที่รับ push จากอุปกรณ์ (กล้อง/edge) ไม่มี session cookie ให้เช็ค — auth เป็น token ที่ฝังใน path เอง (/lpr/:token, /face-push/:token) แทน requireAuth pattern ด้านบน หัวใจของ pattern นี้คือเช็ค token ก่อน parse body เสมอ — ถ้าปล่อยให้ express.raw() parse body ก่อนเช็ค token, request ที่ token ผิด/ไม่มีก็ยัง consume memory เต็มลิมิตอยู่ดี (เพิ่งเป็นบั๊กจริง — ดู LIVE-HIGH-001 ใน REF_security-checklist.md, เจอ flood จริง ~2 req/s ที่ยิงเข้ามาต่อเนื่องก่อนแก้).

// ตัวอย่างจาก src/routes/lpr.js — gate เช็คก่อน rawBodyParser เสมอ
function lprTokenGate(req, res, next) {
  const cam = camByLprToken(req.params.token);
  if (!cam) {
    console.warn('[lpr-push] unknown token — dropped (pre-body)');
    res.status(200).end('OK');  // ตอบโค้ดเดียวกับที่ device จริงคาดหวัง — ไม่บอกใบ้ว่า "token ผิด" ให้คนสุ่มรู้
    req.resume();                // drain socket โดยไม่ buffer เข้า memory
    return;
  }
  req._cam = cam;
  next();
}

app.post('/lpr/:token', lprTokenGate, rawBodyParser, async (req, res) => { ... });

ข้อควรระวัง

  • ตอบ response code เดียวกับที่ device จริงคาดหวังเสมอ (ในเคสนี้คือ 200) แม้ token จะผิด — ไม่ใช่เพราะสุภาพ แต่เพราะเปลี่ยนโค้ดจะทำให้ device ที่ retry-on-error ยิงถี่ขึ้นโดยไม่ตั้งใจ
  • ไม่ต้องตอบ error message ที่บอกว่า "token ผิด" ต่างจาก "ไม่มีข้อมูล" — เป็น oracle ให้คนสุ่ม token ได้
  • ต้อง req.resume() เพื่อ drain socket เมื่อไม่เรียก body parser — ไม่งั้น connection ค้าง

Response Pattern

// รายการ (list)
res.json(rows);

// รายการเดียว
res.json(rows[0]);

// Action result
res.json({ ok: true });

// Error (จาก routeError)
// routeError(res, err, 'endpoint name') → HTTP 500 + { error: '...' }

// Validation error
return res.status(400).json({ error: 'คำอธิบาย' });

// Not found
return res.status(404).json({ error: 'Not found' });
